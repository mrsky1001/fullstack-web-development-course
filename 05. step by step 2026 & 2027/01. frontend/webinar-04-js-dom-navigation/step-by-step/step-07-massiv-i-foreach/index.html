querySelectorAll + forEach = мощь!

// Получаем ВСЕ слайды одной строкой: const slides = document.querySelectorAll('.slide'); const dots = document.querySelectorAll('.dot'); // Функция showSlide — показывает слайд по индексу: function showSlide(index) { slides.forEach(slide => slide.classList.remove('active')); dots.forEach(dot => dot.classList.remove('active')); slides[index].classList.add('active'); dots[index].classList.add('active'); }
