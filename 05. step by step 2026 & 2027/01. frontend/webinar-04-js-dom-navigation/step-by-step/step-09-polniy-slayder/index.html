Полный слайдер: добавляем setInterval

Последний шаг — автопрокрутка! setInterval вызывает nextSlide каждые 3 секунды:

let timerId = null; function nextSlide() { showSlide(currentSlide + 1); } function startAuto() { stopAuto(); // Сбрасываем старый таймер timerId = setInterval(nextSlide, 3000); // ← Вот оно! } function stopAuto() { if (timerId) clearInterval(timerId); } // При клике на стрелку — перейти + перезапустить таймер: rightBtn.addEventListener('click', () => { nextSlide(); startAuto(); }); // Запускаем автопрокрутку сразу: startAuto();
