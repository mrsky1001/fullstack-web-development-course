Проверка совпадения паролей

При регистрации важно проверить, что основной пароль и его повтор совпадают: pass1.value === pass2.value.

Код, который выполняется в этом шаге (откройте main.js):

// Шаг 6: Проверка совпадения паролей
const form = document.getElementById('demoForm');

if (form) {
  form.addEventListener('submit', function (event) {
    event.preventDefault();

    const pass = document.getElementById('userPass');
    const confirm = document.getElementById('userConfirm');

    pass.classList.remove('is-invalid');
    confirm.classList.remove('is-invalid');

    if (!pass.value.trim()) {
      pass.classList.add('is-invalid');
    } else if (pass.value !== confirm.value) {
      confirm.classList.add('is-invalid');
      console.log('Ошибка: пароли не совпадают!');
    } else {
      console.log('Пароли успешно совпали!');
    }
  });
}

Совет: проверьте работу формы в браузере и откройте консоль F12.
